# define paths
BIN_DIR := bin

# links to the main.go file.
CMD_DIR := ./cmd/{{ .ProjectName }}
BINARY_NAME := {{ .ProjectName }}

# basic build cmd
build:
	@mkdir -p $(BIN_DIR)

	go build -o $(BIN_DIR)/$(BINARY_NAME) $(CMD_DIR)

# calls the build cmd, then runs the program
run: build
	./$(BIN_DIR)/$(BINARY_NAME)

test:
	go test ./...
