name: ci

on:
  push:
    branches:
      - main
      - master
  pull_request:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Setup go
        uses: actions/setup-go@v5
        with:
          # tracks whatever version go.mod declares, so bumping the module
          # bumps CI with it
          go-version-file: go.mod
          cache: true

      - name: Vet
        run: go vet ./...

      - name: Build
        run: make build

      - name: Test
        run: make test
