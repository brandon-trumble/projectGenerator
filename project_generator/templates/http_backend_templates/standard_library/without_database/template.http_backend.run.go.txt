package main

import (
	"context"
	"errors"
	"log/slog"
	"net/http"
	"time"

	"{{ .ProjectName }}/internal/repository/memory"
	"{{ .ProjectName }}/internal/service"
)

// run acts as our "main" initialization function.
// it sets up database connections, routes, and gets all the env
// vars(hence the getEnv func)
// bubble all build time errors up to this or run time and handle them in main
// I suggest using slog for these as it's JSON is unmatched
func run(ctx context.Context, getEnv func(string) string) error {
	port := getEnv("PORT")
	if port == "" {
		port = "8080"
	}

	// this project was scaffolded without a database, so users live in memory
	// and are gone on restart. swap this line for a database backed repository
	// when you add one, nothing else has to change.
	users := service.NewUserService(memory.NewUserRepository())

	server := &http.Server{
		Addr:    ":" + port,
		Handler: setupRoutes(users),
	}

	// ctx is cancelled on SIGINT/SIGTERM, which gives in-flight
	// requests a window to finish before the process exits
	go func() {
		<-ctx.Done()

		shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
		defer cancel()

		if err := server.Shutdown(shutdownCtx); err != nil {
			slog.Error("error during server shutdown.", "error", err)
		}
	}()

	slog.Info("server listening.", "port", port)

	if err := server.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
		return err
	}

	return nil
}
