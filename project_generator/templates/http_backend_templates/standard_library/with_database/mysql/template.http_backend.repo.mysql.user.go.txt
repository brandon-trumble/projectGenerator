package mysql

import (
	"context"
	"database/sql"
	"fmt"

	"{{ .ProjectName }}/internal/domain"
)

type UserRepository struct {
	db *sql.DB
}

func NewUserRepository(db *sql.DB) *UserRepository {
	return &UserRepository{db: db}
}

func (r *UserRepository) AllUsers(ctx context.Context) ([]domain.UserModel, error) {
	rows, err := r.db.QueryContext(ctx, `SELECT user_id, user_name, first_name, last_name FROM users`)
	if err != nil {
		return nil, fmt.Errorf("querying users: %w", err)
	}
	defer rows.Close()

	var users []domain.UserModel
	for rows.Next() {
		var user domain.UserModel
		if err := rows.Scan(&user.UserId, &user.UserName, &user.FirstName, &user.LastName); err != nil {
			return nil, fmt.Errorf("scanning user: %w", err)
		}
		users = append(users, user)
	}

	// rows.Err reports errors that ended the loop early, checking it is what
	// separates "no more rows" from "the connection dropped"
	if err := rows.Err(); err != nil {
		return nil, fmt.Errorf("iterating users: %w", err)
	}

	return users, nil
}
